# Bodies for the STNH scripted triggers that STNH's ART depends on.
#
# .docs/architecture/stnh-art.md takes STNH's art and never its common/, but the art wiring IS
# script: the vendored asset_selectors pick clothing with triggers that live in
# STNH's common/scripted_triggers/. Undefined, they are an error.log record
# each; defined `always = no`, every uniform entry is unreachable and every
# leader falls through to vanilla's generic uniform. All of them are declared
# here -- 141 as this file stands; count the top-level declarations rather than
# trusting that figure.
#
# PHASE 3, applied 2026-08-03: the era, identity, role and stratum triggers
# above the INERT divider have real bodies -- 25 of the 141. Everything under
# INERT keeps `always = no`, which is a definition and not a body: it resolves,
# so nothing dangles, and the entry it gates is simply never chosen.
# Reasoning, the era choice and the 4.4 mapping table:
# .docs/decisions/16-phase-3-clothing-triggers.md
#
# Keys are STNH's names, referenced verbatim by the art, so only the FILE takes
# the stg_ prefix. Same exception as decision 10.
#
# Two things that are NOT STNH's definitions, and must not be "corrected" back:
#   * STNH's eras are `years_passed` windows — it is a timeline conversion that
#     starts in the ENT era and advances. STG starts AT TNG/DS9 (.docs/planning/scope.md), so
#     a years_passed body is false at game start and leaves us where we were.
#   * STNH's identity triggers gate on country flags it sets in its own
#     common/. Nothing in STG sets them — the same hole the Breen and Romulan
#     advisor voices had.
#     Species class is the gate here, as it is there.

########## ERA — exactly one may be true PER EMPIRE ##########
#
# "Exactly one" is a constraint on what a single leader sees, not a constant:
# turning on a second era for the SAME empire makes two entries match and the
# first in file order silently wins. Two eras that are true for two DIFFERENT
# empires never meet in one evaluation, so the mirror universe can sit in the
# ENT era while everyone else is in DS9 — which is what the Terran Empire is
# for. Keep any new era mutually exclusive with the two below.

# The DS9-era Starfleet uniform, for everyone except the mirror universe. To
# move the whole mod to TNG S3+ instead, set this to `is_terran_empire = no`
# with tng_clothing_era2 taking this body.
ds9_clothing_era = { is_terran_empire = no }

# STNH's TNG *scenario map*, gated on a global flag we never set. The DS9
# leader entries all require `is_tng_era = no`, so this must stay false.
is_tng_era = { always = no }

# ENT era, TRUE ONLY FOR THE TERRAN EMPIRE. Its own prescripted header says it
# is the Empire of ENT 'In a Mirror, Darkly' and it flies the terran_nx
# shipset, so this is the era its art was cut for. Paired with the two mirror
# identity triggers below; on its own it reaches nothing, because every
# `ent_human_*` row also requires uses_starfleet_uniform, which is FED-only.
enterprise_era = { is_terran_empire = yes }

########## IDENTITY — which empire wears Starfleet ##########

# STNH used six country flags; we use the founder species class. `owner_species`
# in country scope is the form proven by the advisor-voice weights in the
# 2026-08-09 run (zero trigger_impl records).
is_starfleet_uniform_country = {
	exists = owner
	owner = { owner_species = { is_species_class = FED } }
}

is_any_starfleet_empire = { is_starfleet_uniform_country = yes }
is_ufp = { is_starfleet_uniform_country = yes }

# The mirror universe. STG's TER class has exactly one empire, the Terran
# Empire, and STNH cut ten ENT-era mirror uniforms for it that nothing could
# reach: the leader rows want `uses_mirror_starfleet_uniform` and the ruler
# rows want `uses_terran_uniform_ruler`, two different triggers, and both were
# on the INERT list — so every Terran leader fell through the master selector
# to `all/generic_female_uniform_01.dds` while the empire designer showed the
# mirror coat correctly (decision 22's game_setup gate). Same shape as decision
# 22: the scope the designer reads and the scopes the game reads are separate,
# and getting one right says nothing about the other.
is_terran_empire = {
	exists = owner
	owner = { owner_species = { is_species_class = TER } }
}

uses_mirror_starfleet_uniform = { is_terran_empire = yes }
uses_terran_uniform_ruler = { is_terran_empire = yes }

uses_starfleet_uniform = {
	is_starfleet_uniform_country = yes
	uses_clothing_override = no
}

uses_starfleet_uniform_pop = { is_starfleet_uniform_country = yes }

# Non-Federation species wear their own clothes. This is what puts Vulcans,
# Klingons and the rest back in Trek costume, not only humans — but a Vulcan
# serving in the Federation still reads as Starfleet, which is canon.
uses_native_clothes = {
	uses_starfleet_uniform = no
	uses_borg_full_prosthetics = no
	uses_borg_leader_full_prosthetics = no
}

uses_native_clothes_pop = { uses_starfleet_uniform_pop = no }

########## ROLES — 4.4 leader classes and subclasses ##########
#
# 4.4 has four classes (official, commander, scientist, envoy) and subclasses
# as traits. STNH's bodies name `leader_class = legend` and six traits 4.4 no
# longer has, so these are written against vanilla rather than copied.
#
#   commander, no/councilor subclass  -> command   (red)
#   commander + admiral subclass      -> admiral
#   commander + general subclass      -> security  (gold)
#   commander + governor subclass     -> command
#   official                          -> civilian president clothes (see below)
#   scientist                         -> sciences  (blue)
#   envoy                             -> checked directly by the selector

uses_starfleet_admiral = {
	leader_class = commander
	has_trait = subclass_commander_admiral
}

is_sec_ops = {
	leader_class = commander
	has_trait = subclass_commander_general
}

# "hero or admiral" is STNH's name for the plain command uniform; the selector
# entry that uses it also requires `uses_starfleet_admiral = no`, so the
# admiral subclass falls through to the admiral entry rather than matching here.
is_hero_or_admiral = {
	leader_class = commander
	NOR = {
		has_trait = subclass_commander_general
		has_trait = subclass_commander_governor
	}
}

# Literally a military governor. Officials must NOT land here: the entry at
# humanoid_master_male_clothes_01.txt:653 catches `leader_class = official` with
# `is_military_governor = no` and dresses them in human_president_male_*.dds —
# the Federation's civilian administration, which is the canon look and is
# reached only by leaving this false for them.
is_military_governor = {
	leader_class = commander
	has_trait = subclass_commander_governor
}

uses_science_uniform = { leader_class = scientist }

# The native-clothes sections use this trio for non-Federation species.
uses_science_uniform_basic = { leader_class = scientist }
uses_command_uniform = {
	leader_class = commander
	NOR = {
		has_trait = subclass_commander_general
		has_trait = subclass_commander_governor
	}
}
uses_operations_uniform = {
	leader_class = commander
	OR = {
		has_trait = subclass_commander_general
		has_trait = subclass_commander_governor
	}
}

# The Federation President is a civilian; the ruler scope has no Starfleet
# entry, only per-species civilian clothes. Composed rather than `always = yes`
# so that giving either override below a body flips this automatically.
ruler_normal_clothes = {
	uses_federation_president_clothes = no
	uses_starfleet_uniform_ruler = no
}
