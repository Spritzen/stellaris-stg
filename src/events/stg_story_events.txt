############################
#
# Star Trek Galaxies -- the Trek story events.
#
# ROOT = country throughout. stg_story.1 / .2 hang on on_five_year_pulse_country
# and stg_story.30 on on_first_contact, both of which vanilla documents as
# `this = country` in common/on_actions/00_on_actions.txt; the twenty in between
# are rolled out of stg_on_five_year_story_pulse, our own on_action, declared in
# common/on_actions/stg_on_actions.txt.
#
# The delay-then-fire pair is copied from vanilla's action.220 / action.221 in
# events/on_action_events_2.txt. Every reward tier and idiom is copied from
# events/country_events_1.txt and events/envoy_events.txt -- files that were
# opened, per .docs/guides/writing-script.md.
#
# TWO FLAGS ON EVERY STORY EVENT, and they do different jobs:
#   stg_recent_story    10 years, shared -- two popups never land back to back
#   stg_story_told_N    50 years, per event -- the same story never repeats soon
# Both are read in `trigger`, so the pool simply rolls something else, or
# nothing. That is the property `random_events` has and a `mean_time_to_happen`
# does not.
#
# No DLC-gated effect, no `specimen`, no event chain: STG is standalone and has
# to behave the same whatever the host owns.
#
# See .docs/decisions/77-trek-story-events.md
#
############################

namespace = stg_story

# ─────────────────────────────────────────────────────────────────────────────
# The pulse. Vanilla's action.220 / action.221, with our on_action at the end.
# ─────────────────────────────────────────────────────────────────────────────

# Trek story pulse -- set up the delay, so the whole galaxy does not roll on one
# day. `random = 360` is vanilla's own spread on the same five-year pulse.
country_event = {
	id = stg_story.1
	hide_window = yes

	is_triggered_only = yes

	trigger = {
		is_country_type = default
	}

	immediate = {
		country_event = { id = stg_story.2 random = 360 }
	}
}

# Trek story pulse part 2 -- fire the pool.
country_event = {
	id = stg_story.2
	hide_window = yes

	is_triggered_only = yes

	immediate = {
		fire_on_action = { on_action = stg_on_five_year_story_pulse }
	}
}

### The Academy Class (FED)
country_event = {
	id = stg_story.10
	title = "stg_story.10.name"
	desc = "stg_story.10.desc"
	picture = GFX_evt_stg_academy_class
	show_sound = event_administrative_work

	is_triggered_only = yes

	trigger = {
		is_country_type = default
		owner_species = { is_species_class = FED }
		NOR = {
			has_country_flag = stg_recent_story
			has_country_flag = stg_story_told_10
		}
	}

	immediate = {
		set_timed_country_flag = { flag = stg_recent_story days = 3600 }
		set_timed_country_flag = { flag = stg_story_told_10 days = 18000 }
	}

	option = {
		name = "stg_story.10.a"
		add_monthly_resource_mult = {
			resource = society_research
			value = @tier1researchreward
			min = @tier1researchmin
			max = @tier1researchmax
		}
	}
	option = {
		name = "stg_story.10.b"
		add_monthly_resource_mult = {
			resource = physics_research
			value = @tier1researchreward
			min = @tier1researchmin
			max = @tier1researchmax
		}
		random_owned_leader = {
			limit = { leader_class = scientist }
			add_experience = 200
		}
	}
}
