############################
#
# Star Trek Galaxies -- outcome events for the Trek anomaly categories in
# common/anomalies/stg_anomaly_categories.txt.
#
# ROOT = ship scope, FROM = planet scope -- the scopes `anomaly_event` hands a
# ship_event, per common/anomalies/99_README_ANOMALIES.txt.
#
# Every effect, reward tier and guard here is copied from
# events/anomaly_events_1.txt. The `if = { limit = { NOT = { has_deposit_for
# = shipclass_research_station } } clear_deposits = yes }` before a research
# deposit is vanilla's own: without it the planet keeps a deposit that makes the
# research station unbuildable, so the reward is unreachable.
#
# No `specimen = `, no DLC-gated effect: STG is standalone and must behave the
# same whatever the host owns.
#
# See .docs/decisions/75-trek-anomalies.md
#
############################

namespace = stg_anomaly

### Dilithium Matrix -- a crystal seam worth mining
ship_event = {
	id = stg_anomaly.1
	title = "stg_anomaly.1.name"
	desc = "stg_anomaly.1.desc"
	picture = GFX_evt_stg_dilithium_matrix
	show_sound = event_scanner
	location = FROM

	is_triggered_only = yes

	option = {
		name = "stg_anomaly.1.a"
		from = { add_deposit = d_rare_crystals_2 }
		owner = {
			add_monthly_resource_mult = {
				resource = physics_research
				value = @tier1researchreward
				min = @tier1researchmin
				max = @tier1researchmax
			}
		}
	}
}

### Starship Graveyard -- a battle nobody recorded
ship_event = {
	id = stg_anomaly.5
	title = "stg_anomaly.5.name"
	desc = "stg_anomaly.5.desc"
	picture = GFX_evt_stg_starship_graveyard
	show_sound = event_ship_bridge
	location = FROM

	is_triggered_only = yes

	option = {
		name = "stg_anomaly.5.a"
		hidden_effect = {
			FROM = {
				add_modifier = {
					modifier = "ship_graveyard"
					days = -1
				}
				if = {
					limit = {
						NOT = { has_deposit_for = shipclass_research_station }
					}
					clear_deposits = yes
				}
			}
		}
		from = { add_deposit = d_engineering_3 }
	}
}
